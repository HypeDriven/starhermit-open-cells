# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.29.0
engine.minor=31
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# Cards and slots also take the pointer or a finger on the desk: drag a card
# or run onto its target, or tap it and then the target; double-tap sends a
# card home or to a free cell. Every button takes the pointer, touch, or Tab
# and Enter.
control.nav_left=ArrowLeft | Board cursor left
control.nav_right=ArrowRight | Board cursor right
control.nav_up=ArrowUp | Board cursor up
control.nav_down=ArrowDown | Board cursor down
control.confirm=Enter+Space | Select or place
control.undo=KeyZ | Undo
control.hint=KeyH | Hint
control.auto=KeyA | Collect to foundations
control.restart=KeyR | Restart deal
control.camera_reset=Digit0+Numpad0 | Reset camera
control.cancel=Escape | Cancel selection / pause
control.pause=KeyP | Pause
engine.group.nav_left=Board
engine.group.undo=Assists
engine.group.restart=Deal

# Gamepad defaults: the D-pad or left stick moves the board cursor, South
# selects or places, East cancels, Start pauses, West undoes, North hints,
# the right shoulder collects, Select restarts and the right stick resets
# the camera.
engine.gamepad.nav_left=DpadLeft+LeftStickLeft
engine.gamepad.nav_right=DpadRight+LeftStickRight
engine.gamepad.nav_up=DpadUp+LeftStickUp
engine.gamepad.nav_down=DpadDown+LeftStickDown
engine.gamepad.confirm=South
engine.gamepad.cancel=East
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.hint=North
engine.gamepad.auto=RightShoulder
engine.gamepad.restart=Select
engine.gamepad.camera_reset=RightStickPress
